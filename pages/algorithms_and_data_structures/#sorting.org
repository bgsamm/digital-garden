*** Insertion sort

#+begin_src python :results output
  def insertion_sort(array):
      for i in range(1, len(array)):
          key = array[i]
          j = i - 1
          while j >= 0 and key < array[j]:
              array[j + 1] = array[j]
              j -= 1
          array[j + 1] = key
      return array

  print(insertion_sort([5, 2, 4, 6, 1, 3]))
  print(insertion_sort([31, 41, 59, 26, 41, 58]))
#+end_src

#+RESULTS:

  def merge_sort(array):
      if len(array) <= 1:
          return array

      mid = len(array) // 2
      left = array[:mid]
      right = array[mid:]

      merge_sort(left)
      merge_sort(right)

      i = j = 0
      while i < len(left) and j < len(right):
          if left[i] < right[j]:
              array[i + j] = left[i]
              i += 1
          else:
              array[i + j] = right[j]
              j += 1

      while i < len(left):
          array[i + j] = left[i]
          i += 1

      while j < len(right):
          array[i + j] = right[j]
          j += 1

      return array

  print(merge_sort([5, 2, 4, 6, 1, 3]))
  print(merge_sort([31, 41, 59, 26, 41, 58]))
#+end_src

: [1, 2, 3, 4, 5, 6]

  def quick_sort(array):
      def _quick_sort(array, start, end):
          if start >= end:
              return

          pivot = array[start]
          i = start + 1
          j = end
          
          while True:
              while i <= end and array[i] <= pivot:
                  i += 1
              while j > start and array[j] >= pivot:
                  j -= 1

              if j < i:
                  break

              array[i], array[j] = array[j], array[i]

          array[start], array[j] = array[j], array[start]

          _quick_sort(array, start, j - 1)
          _quick_sort(array, j + 1, end)

      _quick_sort(array, 0, len(array) - 1)
      return array

  print(quick_sort([5, 2, 4, 6, 1, 3]))
  print(quick_sort([31, 41, 59, 26, 41, 58]))
#+end_src

: [1, 2, 3, 4, 5, 6]
: [26, 31, 41, 41, 58, 59]
